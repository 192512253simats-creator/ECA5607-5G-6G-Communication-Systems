clc;
clear;
close all;

% ECA56 Experiment 1
% FDMA, TDMA and CDMA comparison
% ASCII-only MATLAB code for compatibility

%% OBJECTIVE 1: Estimate and compare user capacity

% System parameters
 totalBandwidthHz = 1e6;       % Total bandwidth: 1 MHz
 fdmaChannelHz = 200e3;        % FDMA channel width: 200 kHz
 tdmaSlots = 8;                % TDMA slots per frame
 processingGain = 64;          % CDMA processing gain
 targetSIRdB = 7;              % Target CDMA SIR in dB

% Capacity estimates
fdmaUsers = floor(totalBandwidthHz / fdmaChannelHz);
tdmaUsers = tdmaSlots;
targetSIR = 10^(targetSIRdB / 10);
cdmaUsers = floor(1 + processingGain / targetSIR);

fprintf('\nOBJECTIVE 1: CAPACITY COMPARISON\n');
fprintf('Total bandwidth: %.0f kHz\n', totalBandwidthHz / 1e3);
fprintf('FDMA channel width: %.0f kHz\n', fdmaChannelHz / 1e3);
fprintf('TDMA slots: %d\n', tdmaSlots);
fprintf('CDMA processing gain: %d\n', processingGain);
fprintf('Target CDMA SIR: %.1f dB\n\n', targetSIRdB);
fprintf('FDMA: %d illustrative users\n', fdmaUsers);
fprintf('TDMA: %d illustrative users\n', tdmaUsers);
fprintf('CDMA: %d illustrative users\n\n', cdmaUsers);

method = {'FDMA'; 'TDMA'; 'CDMA'};
supportedUsers = [fdmaUsers; tdmaUsers; cdmaUsers];
capacityTable = table(method, supportedUsers, ...
    'VariableNames', {'Method', 'SupportedUsers'});
disp(capacityTable);

figure;
bar(supportedUsers);
set(gca, 'XTick', 1:3, 'XTickLabel', method);
xlabel('Multiple Access Method');
ylabel('Illustrative Number of Users');
title('FDMA, TDMA and CDMA Capacity Estimate');
grid on;

%% OBJECTIVE 2: Analyze CDMA SIR as user count increases

K = 2:30;
sirLinear = processingGain ./ (K - 1);
sirDb = 10 * log10(sirLinear);

fprintf('OBJECTIVE 2: CDMA SIR ANALYSIS\n');
fprintf('The estimated SIR decreases as the number of users increases.\n');
fprintf('The target SIR is %.1f dB.\n\n', targetSIRdB);

sirTable = table(K', sirDb', ...
    'VariableNames', {'NumberOfUsers', 'EstimatedSIR_dB'});
disp(sirTable);

figure;
plot(K, sirDb, 'o-', 'LineWidth', 1.5);
hold on;
plot([min(K) max(K)], [targetSIRdB targetSIRdB], 'r--', ...
    'LineWidth', 1.5);
xlabel('Number of Simultaneous CDMA Users');
ylabel('Estimated SIR (dB)');
title('CDMA SIR versus Number of Users');
legend('Estimated SIR', 'Target SIR', 'Location', 'best');
grid on;
hold off;

%% OBJECTIVE 3: Compare access methods across mobile generations

generation = {'1G'; '2G'; '2G'; '3G'; '4G'; '5G'};
systemName = {'AMPS'; 'GSM'; 'IS-95'; 'UMTS'; 'LTE'; '5G NR'};
accessMethod = {'FDMA'; 'TDMA'; 'CDMA'; 'WCDMA'; 'OFDMA'; 'OFDMA-based'};
keyIdea = {'Separate frequency channels'; ...
           'Separate time slots'; ...
           'Code-based user separation'; ...
           'Wideband code-based access'; ...
           'Orthogonal frequency resources'; ...
           'Flexible OFDM-based radio access'};

evolutionTable = table(generation, systemName, accessMethod, keyIdea, ...
    'VariableNames', {'Generation', 'System', 'AccessMethod', 'KeyIdea'});

fprintf('\nOBJECTIVE 3: MOBILE GENERATION COMPARISON\n');
disp(evolutionTable);

%% OBJECTIVE 4: Draw resource-allocation diagrams

figure;
colors = lines(4);

% FDMA: each user has a different frequency channel
subplot(1, 3, 1);
hold on;
for u = 1:4
    rectangle('Position', [0, u-1, 1, 0.8], ...
        'FaceColor', colors(u,:), 'EdgeColor', 'k');
    text(0.5, u-0.6, sprintf('User %d', u), ...
        'HorizontalAlignment', 'center');
end
xlim([0 1]);
ylim([0 4]);
xlabel('Time');
ylabel('Frequency Channel');
title('FDMA');
set(gca, 'YTick', []);
hold off;

% TDMA: each user has a different time slot
subplot(1, 3, 2);
hold on;
for u = 1:4
    rectangle('Position', [u-1, 0, 0.8, 1], ...
        'FaceColor', colors(u,:), 'EdgeColor', 'k');
    text(u-0.6, 0.5, sprintf('User %d', u), ...
        'HorizontalAlignment', 'center');
end
xlim([0 4]);
ylim([0 1]);
xlabel('Time Slot');
ylabel('Frequency Channel');
title('TDMA');
set(gca, 'YTick', []);
hold off;

% CDMA: users share time and frequency but use different codes
subplot(1, 3, 3);
hold on;
for u = 1:4
    rectangle('Position', [0, 0, 1, 1], ...
        'FaceColor', colors(u,:), 'FaceAlpha', 0.18, ...
        'EdgeColor', colors(u,:), 'LineWidth', 1.5);
end
text(0.5, 0.62, 'Users 1 to 4 share', ...
    'HorizontalAlignment', 'center');
text(0.5, 0.40, 'band; different codes', ...
    'HorizontalAlignment', 'center');
xlim([0 1]);
ylim([0 1]);
xlabel('Time');
ylabel('Frequency Band');
title('CDMA');
set(gca, 'YTick', []);
hold off;

sgtitle('Conceptual Time-Frequency Resource Allocation');

fprintf('\nAll four objectives have finished running.\n');
fprintf('Note: These are simplified illustrative models, not real network guarantees.\n');
